"""CircuitPython inline bootstrap — sent through raw REPL.

Registers library modules via the class-as-module pattern (CircuitPython
lacks types.ModuleType), then execs the test file and runs it through
the harness.

Generated by circuitpython_bootstrap.py.  The markers below are
replaced at generation time:

- $STUB_REGISTRATIONS  — _register_stub() calls for each module
- $MODULE_POPULATIONS  — _populate_module() calls wrapped in try/except
- $TEST_SOURCE         — escaped test file source
- $FILTER_REPR         — name_filter argument for run_module
"""

import sys


def _register_stub(module_name):
    """Register an empty class as a module placeholder in sys.modules."""
    parts = module_name.rsplit('.', 1)
    package = parts[0] if len(parts) > 1 else module_name

    class _Mod:
        pass

    _Mod.__name__ = module_name
    _Mod.__package__ = package
    sys.modules[module_name] = _Mod
    if len(parts) > 1:
        parent = parts[0]
        attr = parts[1]
        if parent in sys.modules:
            setattr(sys.modules[parent], attr, _Mod)


def _populate_module(module_name, source_code):
    """Exec source into the stub already registered in sys.modules."""
    parts = module_name.rsplit('.', 1)
    package = parts[0] if len(parts) > 1 else module_name
    namespace = {'__name__': module_name, '__package__': package}
    exec(source_code, namespace)
    module_obj = sys.modules[module_name]
    for _key in namespace:
        if not _key.startswith('__'):
            setattr(module_obj, _key, namespace[_key])
    if len(parts) > 1:
        parent = parts[0]
        attr = parts[1]
        if parent in sys.modules:
            setattr(sys.modules[parent], attr, module_obj)


# Pass 1: register empty stubs so import statements resolve.
$STUB_REGISTRATIONS

# Pass 2: populate modules; defer any with unresolved forward references.
_deferred = []
$MODULE_POPULATIONS

# Pass 3: retry deferred modules (forward references now resolved).
for _mod_name, _mod_src in _deferred:
    _populate_module(_mod_name, _mod_src)

# Execute the test file.
_test_namespace = {}
exec($TEST_SOURCE, _test_namespace)

# Convert namespace to attribute object for run_module.
class _TestModule:
    pass

for _key in _test_namespace:
    setattr(_TestModule, _key, _test_namespace[_key])

# Run.
from chumicro_test_harness.runner import run_module

run_module(_TestModule, name_filter=$FILTER_REPR)
